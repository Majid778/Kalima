    **ArabicText Component:**

    3. Create src/components/ArabicText.tsx:
       ```typescript
       import React from 'react';
       import { Text, StyleSheet, TextStyle, TextProps } from 'react-native';
       import { fonts, arabicFontSizes, arabicLineHeights, ArabicFontSize } from '../theme/typography';
       import { colors } from '../theme/colors';

       interface ArabicTextProps extends Omit<TextProps, 'style'> {
         children: string;
         style?: TextStyle;
         size?: ArabicFontSize;
         color?: string;
       }

       export function ArabicText({
         children,
         style,
         size = 'medium',
         color,
         ...textProps
       }: ArabicTextProps) {
         return (
           <Text
             style={[
               styles.arabic,
               {
                 fontSize: arabicFontSizes[size],
                 lineHeight: arabicLineHeights[size],
               },
               color ? { color } : null,
               style,
             ]}
             allowFontScaling={false}
             {...textProps}
           >
             {children}
           </Text>
         );
       }

       const styles = StyleSheet.create({
         arabic: {
           fontFamily: fonts.arabic,     // 'KFGQPC-Uthmanic-Script-HAFS' per D-03
           writingDirection: 'rtl',       // Per-component RTL only (D-10, ARAB-02)
           textAlign: 'right',            // D-10
           color: colors.text,            // Default white; override via color prop
         },
       });
       ```

       KEY IMPLEMENTATION NOTES:
       - `allowFontScaling={false}` prevents OS accessibility scaling from breaking Arabic layout. Arabic already uses large font sizes (24-42pt).
       - `writingDirection: 'rtl'` is set in the StyleSheet, NOT via I18nManager.forceRTL() (D-10)
       - `textAlign: 'right'` ensures Arabic text aligns to the right side of its container
       - The `color` prop defaults to `colors.text` (#FFFFFF) but can be overridden for green accent text (#4ade80)
       - The component extends TextProps (minus style) so it supports `numberOfLines`, `onPress`, `testID`, etc.
