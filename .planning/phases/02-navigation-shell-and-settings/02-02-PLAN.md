```typescript
import React from 'react';
import { View, TouchableOpacity, Text, StyleSheet } from 'react-native';
import { colors } from '../theme/colors';

interface Segment<T extends string> {
  label: string;
  value: T;
}

interface SegmentedControlProps<T extends string> {
  segments: Segment<T>[];
  selectedValue: T;
  onValueChange: (value: T) => void;
}

export function SegmentedControl<T extends string>({
  segments, selectedValue, onValueChange,
}: SegmentedControlProps<T>) {
  return (
    <View style={styles.container}>
      {segments.map((segment) => {
        const isSelected = segment.value === selectedValue;
        return (
          <TouchableOpacity
            key={segment.value}
            style={[styles.segment, isSelected && styles.segmentSelected]}
            onPress={() => onValueChange(segment.value)}
            accessibilityRole="button"
            accessibilityState={{ selected: isSelected }}
          >
            <Text style={[styles.label, isSelected && styles.labelSelected]}>
              {segment.label}
            </Text>
          </TouchableOpacity>
        );
      })}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flexDirection: 'row',
    borderRadius: 8,
    overflow: 'hidden',
  },
  segment: {
    flex: 1,
    paddingVertical: 8,
    alignItems: 'center',
    backgroundColor: colors.surfaceLight,    // #2a2a2a per D-09
  },
  segmentSelected: {
    backgroundColor: colors.primary,          // #4ade80 per D-09
  },
  label: {
    color: colors.textMuted,                  // muted text per D-09
    fontSize: 14,
    fontWeight: '600',
  },
  labelSelected: {
    color: '#FFFFFF',                          // white text per D-09
  },
});
```

```typescript
import React from 'react';
import { View, Text, Switch, StyleSheet, SafeAreaView } from 'react-native';
import { colors } from '../theme/colors';
import { useSettingsStore } from '../stores/settingsStore';
import { SegmentedControl } from '../components/SegmentedControl';
import { ArabicFontSize } from '../theme/typography';

const FONT_SIZE_SEGMENTS: { label: string; value: ArabicFontSize }[] = [
  { label: 'S', value: 'small' },
  { label: 'M', value: 'medium' },
  { label: 'L', value: 'large' },
];

export function ProfileScreen() {
  const arabicFontSize = useSettingsStore((s) => s.arabicFontSize);
  const showTransliteration = useSettingsStore((s) => s.showTransliteration);
  const setArabicFontSize = useSettingsStore((s) => s.setArabicFontSize);
  const setShowTransliteration = useSettingsStore((s) => s.setShowTransliteration);

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.content}>
        {/* Section header per D-06 */}
        <Text style={styles.sectionHeader}>Display</Text>

        {/* Arabic Font Size row per D-07, D-08 */}
        <View style={styles.row}>
          <Text style={styles.rowLabel}>Arabic Font Size</Text>
          <SegmentedControl
            segments={FONT_SIZE_SEGMENTS}
            selectedValue={arabicFontSize}
            onValueChange={setArabicFontSize}
          />
        </View>

        {/* Transliteration toggle per D-07, D-13, D-14 */}
        <View style={styles.row}>
          <Text style={styles.rowLabel}>Show Transliteration</Text>
          <Switch
            value={showTransliteration}
            onValueChange={setShowTransliteration}
            trackColor={{ false: '#3e3e3e', true: colors.primary }}
            thumbColor="#FFFFFF"
          />
        </View>
      </View>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: colors.background,       // #0a0a0a
  },
  content: {
    padding: 20,
  },
  sectionHeader: {
    color: colors.textSecondary,              // #a1a1aa
    fontSize: 13,
    fontWeight: '600',
    textTransform: 'uppercase',
    letterSpacing: 1,
    marginBottom: 16,

  row: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    alignItems: 'center',
    paddingVertical: 14,
    borderBottomWidth: StyleSheet.hairlineWidth,
    borderBottomColor: colors.border,          // #27272a
  },
  rowLabel: {
    color: colors.text,                        // #FFFFFF
    fontSize: 16,
  },
});
```

1. Modify `src/components/ArabicText.tsx` — make the `size` prop optional and fall back to the global settings store value when not provided (per D-10, Pattern 4 from research):

Change the destructuring from `size = 'medium'` to `size` (no default). Add:
```typescript
import { useSettingsStore } from '../stores/settingsStore';
```

Inside the component function, before the return:
```typescript
const globalSize = useSettingsStore((s) => s.arabicFontSize);
const effectiveSize = size ?? globalSize;
```

Then use `effectiveSize` everywhere `size` was used:
```typescript
fontSize: arabicFontSizes[effectiveSize],
lineHeight: arabicLineHeights[effectiveSize],
```

This means:
- If `size` prop is explicitly passed (e.g., `<ArabicText size="large">`), it overrides the store
- If `size` prop is omitted (e.g., `<ArabicText>`), it reads from the settings store
- Backward compatible: existing callers with explicit `size` props are unaffected
